# `@bogdanrn/yt-embed` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a fresh, MIT-licensed npm package that wraps the YouTube IFrame Player API in a Promise-based, `EventTarget`-based, TypeScript-strict class.

**Architecture:** A single class `YTEmbed extends EventTarget` with auto-queueing Promise-returning method wrappers, DOM-style events, lazy-attached extensions, and a code-generated method/event surface driven by `@types/youtube`. Module-private cached iframe-API loader. Side-effect-free import (SSR-safe).

**Tech Stack:** TypeScript strict, `tsdown` (Rolldown) for bundling, `vitest` + `jsdom` for tests, Biome for lint+format, `ts-morph` for codegen, `typedoc` + `typedoc-plugin-markdown` for docs, `changesets` for release, `pnpm` package manager, `gh` CLI for repo bootstrap.

Each `src/` module has a single responsibility:
- `loadIframeApi.ts` — script injection + cached promise. No DOM access at module top.
- `playerState.ts` — `PlayerState` constants, `PlayerStateCode` type. Pure data.
- `errors.ts` — three custom error classes. Pure declarations.
- `types.ts` — `YTEmbedOptions`, `YTEmbedEventMap`, `MethodCallOptions`, type aliases. Pure declarations.
- `listenerTracker.ts` — small helper that maintains per-event-type listener counts and fires callbacks on first-add / last-remove. Used by `YTEmbed` to drive lazy extension attach/detach.

- `extensions/volumeChange.ts` — the built-in extension.
- `YTEmbed.ts` — the main class. Uses everything above.
- `index.ts` — barrel only; no logic.

### Task 1.1: Author `scripts/generate.ts` (functionNames only)

- [ ] **Step 1: Write the script**

```ts
#!/usr/bin/env tsx
import { writeFileSync } from 'node:fs';
import { Project } from 'ts-morph';

const project = new Project({ tsConfigFilePath: 'tsconfig.json' });

const ytNamespace = project.getSourceFiles()
  .flatMap((f) => f.getModules())
  .find((m) => m.getName() === 'YT');

if (!ytNamespace) {
  throw new Error(
    'YT global namespace not found. Is @types/youtube installed and listed in tsconfig "types"?',
  );
}

const playerInterface = ytNamespace.getInterface('Player');
if (!playerInterface) {
  throw new Error('Interface YT.Player not found in @types/youtube');
}

const methodNames = playerInterface
  .getMethods()
  .map((m) => m.getName())
  .sort();

writeFileSync(
  'src/functionNames.generated.ts',
  `// AUTO-GENERATED from @types/youtube. Do not edit.\n` +
    `// Regenerate with \`pnpm generate\`.\n\n` +
    `export const functionNames = ${JSON.stringify(methodNames, null, 2)} as const;\n\n` +
    `export type FunctionName = (typeof functionNames)[number];\n`,
);

console.log(`Wrote src/functionNames.generated.ts (${methodNames.length} methods)`);
```

```ts
import { IframeApiLoadError } from './errors.js';

const SCRIPT_URL = 'https://www.youtube.com/iframe_api';

let cachedPromise: Promise<typeof YT> | null = null;

export interface YTEmbedConfig {
  scriptUrl?: string;
}
export const config: YTEmbedConfig = {};

export function loadIframeApi(): Promise<typeof YT> {
  if (cachedPromise) return cachedPromise;

  cachedPromise = new Promise<typeof YT>((resolve, reject) => {
    // biome-ignore lint/suspicious/noExplicitAny: window.YT augmentation is global.
    const existingYT = (window as any).YT;
    if (existingYT?.Player) {
      resolve(existingYT);
      return;
    }

    // biome-ignore lint/suspicious/noExplicitAny: window callback augmentation.
    const previous = (window as any).onYouTubeIframeAPIReady as (() => void) | undefined;
    // biome-ignore lint/suspicious/noExplicitAny: window callback augmentation.
    (window as any).onYouTubeIframeAPIReady = () => {
      previous?.();
      // biome-ignore lint/suspicious/noExplicitAny: window.YT global.
      resolve((window as any).YT as typeof YT);
    };

    const url = config.scriptUrl ?? SCRIPT_URL;
    const script = document.createElement('script');
    script.src = url;
    script.async = true;
    script.addEventListener('error', (event) => {
      cachedPromise = null; // allow retry after failure
      reject(new IframeApiLoadError('Failed to load YouTube IFrame API', { cause: event }));
    });
    document.head.appendChild(script);
  });

  return cachedPromise;
}

  cachedPromise = null;
```

```ts
export interface ListenerTrackerHooks {
  onFirstAdd: (type: string) => void;
  onLastRemove: (type: string) => void;
}

export class ListenerTracker {
  readonly #counts = new Map<string, number>();
  readonly #hooks: ListenerTrackerHooks;

  constructor(hooks: ListenerTrackerHooks) {
    this.#hooks = hooks;
  }

  add(type: string): void {
    const next = (this.#counts.get(type) ?? 0) + 1;
    this.#counts.set(type, next);
    if (next === 1) this.#hooks.onFirstAdd(type);
  }

  remove(type: string): void {
    const current = this.#counts.get(type) ?? 0;
    if (current <= 0) return;
    const next = current - 1;
    if (next === 0) {
      this.#counts.delete(type);
      this.#hooks.onLastRemove(type);
    } else {
      this.#counts.set(type, next);
    }
  }

  count(type: string): number {
    return this.#counts.get(type) ?? 0;
  }
}
```

```ts
type Pending = {
  run: () => void;
  reject: (e: unknown) => void;
};
#queue: Pending[] = [];

#flushQueue(): void {
  while (this.#queue.length) {
    const next = this.#queue.shift();
    next?.run();
  }
}

#assertNotDestroyed(): void {
  if (this.#destroyed) {
    throw new PlayerDestroyedError('Player has been destroyed');
  }
}

call<K extends FunctionName>(method: K, ...args: unknown[]): Promise<unknown> {
  return this.#exec(method as string, args, undefined);
}

// Overload to support trailing options object:
//   call('playVideo', { awaitState: true })
//   call('seekTo', 30, true, { awaitState: true })
// Detect the trailing arg shape at runtime.
#exec(method: string, rawArgs: unknown[], _options?: MethodCallOptions): Promise<unknown> {
  const lastArg = rawArgs[rawArgs.length - 1];
  const options =
    lastArg && typeof lastArg === 'object' && !Array.isArray(lastArg) &&
    ('awaitState' in (lastArg as object) || 'signal' in (lastArg as object))
      ? (rawArgs.pop() as MethodCallOptions)
      : undefined;

  return new Promise((resolve, reject) => {
    options?.signal?.throwIfAborted?.();

    const onAbort = () => reject(options?.signal?.reason ?? new DOMException('Aborted', 'AbortError'));
    options?.signal?.addEventListener('abort', onAbort, { once: true });

    const run = () => {
      if (this.#destroyed) {
        reject(new PlayerDestroyedError('Player destroyed before call'));
        return;
      }
      try {
        const player = this.#player as unknown as Record<string, (...a: unknown[]) => unknown>;
        const result = player[method]?.(...rawArgs);
        if (options?.awaitState) {
          const onState = (e: Event) => {
            this.removeEventListener('statechange', onState);
            options.signal?.removeEventListener('abort', onAbort);
            void e;
            resolve(result);
          };
          this.addEventListener('statechange', onState as EventListener);
        } else {
          options?.signal?.removeEventListener('abort', onAbort);
          resolve(result);
        }
      } catch (err) {
        reject(err);
      }
    };

    if (this.#readyResolved) {
      run();
    } else {
      this.#queue.push({ run, reject });
    }
  });
}
```

```ts
import type { Extension } from './types.js';
import type { YTEmbed } from '../YTEmbed.js';

export interface VolumeChangeExtensionOptions {
  /** Polling interval in ms. Default: 250. */
  interval?: number;
}

export function volumeChangeExtension(
  options: VolumeChangeExtensionOptions = {},
): Extension {
  const interval = options.interval ?? 250;

  return {
    events: ['volumechange'],
    attach(player: YTEmbed) {
      let last: { volume: number; muted: boolean } | null = null;

      async function readAndMaybeEmit(): Promise<void> {
        const [volume, muted] = await Promise.all([
          (player as unknown as { getVolume: () => Promise<number> }).getVolume(),
          (player as unknown as { isMuted: () => Promise<boolean> }).isMuted(),
        ]);
        if (last === null || last.volume !== volume || last.muted !== muted) {
          last = { volume, muted };
          player.dispatchEvent(new CustomEvent('volumechange', { detail: last }));
        }
      }

      // Hook setters by listening on the player itself for our own method
      // resolution. Easiest signal: re-read after our setVolume/mute/unMute
      // resolve. We use a wrapping event-listener on a custom 'method:done'
      // event, OR — simpler — we just poll AND read-after-set via promise.
      // The simplest path: poll on a timer.
      const timer = setInterval(() => void readAndMaybeEmit(), interval);

      // Hook path: after each volume-affecting method resolves, force a read.
      const hookForMethod = async <T>(promise: Promise<T>): Promise<T> => {
        const result = await promise;
        await readAndMaybeEmit();
        return result;
      };
      // biome-ignore lint/suspicious/noExplicitAny: monkey-patch own instance methods.
      const p = player as unknown as Record<string, (...a: unknown[]) => Promise<unknown>>;
      const originals = {
        setVolume: p.setVolume.bind(player),
        mute: p.mute.bind(player),
        unMute: p.unMute.bind(player),
      };
      p.setVolume = (...a: unknown[]) => hookForMethod(originals.setVolume(...a));
      p.mute = (...a: unknown[]) => hookForMethod(originals.mute(...a));
      p.unMute = (...a: unknown[]) => hookForMethod(originals.unMute(...a));

      return () => {
        clearInterval(timer);
        p.setVolume = originals.setVolume;
        p.mute = originals.mute;
        p.unMute = originals.unMute;
      };
    },
  };
}
```

```tsx
import { StrictMode, useEffect, useRef, useState } from 'react';
import { createRoot } from 'react-dom/client';
import { YTEmbed, PlayerState, type PlayerStateCode, volumeChangeExtension } from '@bogdanrn/yt-embed';

function YouTube({ videoId }: { videoId: string }) {
  const ref = useRef<HTMLDivElement>(null);
  const [state, setState] = useState<PlayerStateCode>(-1);
  const [volume, setVolume] = useState<number>(0);

  useEffect(() => {
    if (!ref.current) return;
    const ac = new AbortController();
    const player = new YTEmbed(ref.current, {
      videoId,
      signal: ac.signal,
      extensions: [volumeChangeExtension()],
    });
    player.addEventListener('statechange', (e) => setState(e.detail.state), { signal: ac.signal });
    player.addEventListener('volumechange', (e) => setVolume(e.detail.volume), { signal: ac.signal });
    return () => {
      ac.abort();
      player.destroy();
    };
  }, [videoId]);

  return (
    <div>
      <div ref={ref} />
      <p>state: {state} (PLAYING={PlayerState.PLAYING})</p>
      <p>volume: {volume}</p>
    </div>
  );
}

createRoot(document.getElementById('root')!).render(
  <StrictMode>
    <YouTube videoId="M7lc1UVf-VE" />
  </StrictMode>,
```

```ts
#!/usr/bin/env tsx
import { readFileSync, writeFileSync } from 'node:fs';

function inject(readme: string, marker: string, contentPath: string): string {
  const content = readFileSync(contentPath, 'utf8')
    .split('\n')
    .filter((l) => !l.startsWith('<!-- AUTO-GENERATED'))
    .join('\n')
    .trim();
  const start = `<!-- ${marker}:start -->`;
  const end = `<!-- ${marker}:end -->`;
  const re = new RegExp(`${start}[\\s\\S]*?${end}`);
  return readme.replace(re, `${start}\n${content}\n${end}`);
}

let readme = readFileSync('README.md', 'utf8');
readme = inject(readme, 'methods', 'docs/methods.md');
readme = inject(readme, 'events', 'docs/events.md');
writeFileSync('README.md', readme);
console.log('README updated.');
```
